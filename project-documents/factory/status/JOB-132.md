```python
import json
from pathlib import Path

f = json.loads(Path('visual-assets/v10_1/legacy/fixtures.json').read_text())
errors = []
def check(ok, message):
    if not ok:
        errors.append(message)
def integer(v):
    return type(v) is int
def season_winner(a, b):
    ka = (a['score'], -a['leaguePosition'], a['leaguePoints'])
    kb = (b['score'], -b['leaguePosition'], b['leaguePoints'])
    return 'daniel' if ka > kb else 'nik' if kb > ka else 'draw'
teams = dict(premier_league=20, laliga=20, bundesliga=18, serie_a=20, ligue_1=20)
evidence = f['fixtureEvidence']['showdowns']
checked_keys = set()
summary = []
counts = dict(cards=0, seasons=0, scores=0, trophies=0, margins=0)
for name, frame in f['frames'].items():
    check(frame['previewLabel'] == 'Preview data', f'{name}: preview label')
    check(frame['status'] in ('loading','empty','unavailable','partial','ready'), f'{name}: status')
    if frame['status'] in ('loading','empty','unavailable'):
        check(frame['showdowns'] == [], f'{name}: failed/empty/loading read must not invent data')
    if frame['status'] == 'partial':
        c = frame['coverage']
        check(integer(c['readable']) and integer(c['indexed']) and 0 <= c['readable'] < c['indexed'], f'{name}: coverage bounds')
        check(c['indexed'] == len(frame['showdowns']), f'{name}: indexed coverage')
        check(c['readable'] == sum(s['status'] != 'unavailable' for s in frame['showdowns']), f'{name}: readable coverage')
    if 'interimLabel' in frame:
        check(frame['interimLabel'] == 'Current Showdown only. Career history is not yet available.', f'{name}: interim label')
    check(len({s['number'] for s in frame['showdowns']}) == len(frame['showdowns']), f'{name}: duplicate number')
    for sd in frame['showdowns']:
        counts['cards'] += 1
        label = f"{name}/#{sd['number']}"
        check(integer(sd['number']) and sd['number'] > 0, label+': number')
        check(sd['status'] in ('completed','in-progress','completion-pending','abandoned','unavailable'), label+': card status')
        if sd['status'] in ('abandoned','unavailable'):
            check(set(sd) == {'number','status'}, label+': status-only row has data')
            continue
        check(sd['leagueId'] in teams, label+': league id')
        n = teams[sd['leagueId']]
        check(sd['totalSeasons'] in (1,3,5,10) and integer(sd['totalSeasons']), label+': configured season count')
        check(integer(sd['seasonsPlayed']) and 0 <= sd['seasonsPlayed'] <= sd['totalSeasons'], label+': played seasons bounds')
        check(sd['seasonsPlayed'] == len(sd['seasons']), label+': season count')
        check([s['season'] for s in sd['seasons']] == list(range(1,sd['seasonsPlayed']+1)), label+': season numbering')
        if sd['status'] in ('completed','completion-pending'):
            check(sd['seasonsPlayed'] == sd['totalSeasons'], label+': final result incomplete')
        check(list(sd['clubs']) == ['daniel','nik'] and all(sd['clubs'].values()), label+': club order')
        check(list(sd['totals']) == ['daniel','nik'], label+': total order')
        key = f"{sd['status']}:{sd['number']}"
        proof = evidence[key]
        check(len(proof['seasons']) == len(sd['seasons']), label+': evidence length')
        totals = dict(daniel=0,nik=0)
        trophy = {m:dict(championsLeague=0,leagueTitles=0,domesticCups=0,total=0) for m in totals}
        for season, raw in zip(sd['seasons'],proof['seasons']):
            counts['seasons'] += 1
            sl = label+f"/season{season['season']}"
            check(raw['season'] == season['season'], sl+': evidence season')
            check(season['leaguePosition']['daniel'] != season['leaguePosition']['nik'], sl+': shared league position')
            per_manager = {}
            check(list(raw['inputs']) == ['daniel','nik'], sl+': evidence order')
            for m in ('daniel','nik'):
                inp = raw['inputs'][m]
                for field, upper, lower in (('leaguePosition',n,1),('leaguePoints',(n-1)*6,0),('leagueGoals',300,0)):
                    check(integer(inp[field]) and lower <= inp[field] <= upper, sl+'/'+m+': '+field+' bounds')
                    check(inp[field] == season[field][m], sl+'/'+m+': '+field+' mismatch')
                    check(list(season[field]) == ['daniel','nik'], sl+': '+field+' manager order')
                for field in ('domesticCup','championsLeague','topScorer','topAssist'):
                    check(type(inp[field]) is bool, sl+'/'+m+': '+field+' boolean')
                breakdown = dict(championsLeague=5*int(inp['championsLeague']),
                                 leagueTitle=3*int(inp['leaguePosition']==1),
                                 domesticCup=int(inp['domesticCup']),
                                 performanceBonus=int(inp['leaguePoints']>=100 or inp['leagueGoals']>=100),
                                 awardsBonus=int(inp['topScorer'] or inp['topAssist']))
                check(breakdown == raw['expectedBreakdown'][m], sl+'/'+m+': breakdown mismatch')
                score = sum(breakdown.values())
                check(integer(season['score'][m]) and 0 <= score <= 11 and score == season['score'][m], sl+'/'+m+': score mismatch')
                check(list(season['score']) == ['daniel','nik'], sl+': score order')
                totals[m] += score
                per_manager[m] = dict(inp,score=score)
                trophy[m]['championsLeague'] += int(inp['championsLeague'])
                trophy[m]['leagueTitles'] += int(inp['leaguePosition']==1)
                trophy[m]['domesticCups'] += int(inp['domesticCup'])
                trophy[m]['total'] = sum(trophy[m][k] for k in ('championsLeague','leagueTitles','domesticCups'))
                counts['scores'] += 1
            check(season['winner'] == season_winner(per_manager['daniel'],per_manager['nik']), sl+': season winner')
        check(totals == sd['totals'], label+': total mismatch')
        check(all(integer(v) and 0 <= v <= sd['seasonsPlayed']*11 for v in sd['totals'].values()), label+': total bounds/types')
        winner = 'daniel' if totals['daniel'] > totals['nik'] else 'nik' if totals['nik'] > totals['daniel'] else 'draw'
        if sd['status'] in ('completed','completion-pending'):
            check(sd['winner'] == winner, label+': final winner')
        else:
            check('winner' not in sd, label+': active outcome is not final')
        check(trophy == proof['expectedTrophies'], label+': trophy counts')
        check(list(proof['expectedTrophies']) == ['daniel','nik'], label+': trophy order')
        for m, cabinet in proof['expectedTrophies'].items():
            for field, value in cabinet.items():
                bound = sd['seasonsPlayed'] * (3 if field == 'total' else 1)
                check(integer(value) and 0 <= value <= bound, label+'/'+m+': trophy type/bounds')
        check(integer(proof['expectedMargin']) and proof['expectedMargin'] == abs(totals['daniel']-totals['nik']), label+': margin')
        counts['trophies'] += 8
        counts['margins'] += 1
        if key not in checked_keys:
            summary.append(f"{key}: totals {totals['daniel']}:{totals['nik']}; {'leader' if sd['status']=='in-progress' else 'winner'} {winner}; margin {proof['expectedMargin']}; trophies {trophy['daniel']['total']}:{trophy['nik']['total']}")
        checked_keys.add(key)
check(checked_keys == set(evidence), 'Unused or missing scoring evidence')
for k in ('loading','partial','unavailable'):
    check(f['strings']['states'][k] != k.upper(), 'Raw status token displayed: '+k)
    check(f['stringMetadata']['states.'+k]['source'] == 'new', 'New-copy provenance: '+k)
check(f['stringMetadata']['tagline']['kind'] == 'decorative-brand-text', 'Tagline classification')
print('Legacy fixture recomputation:')
print('\n'.join(summary))
print(f"Frames: {len(f['frames'])}; cards: {counts['cards']}; unique scoring scenarios: {len(checked_keys)}")
print(f"Season occurrences: {counts['seasons']}; computed manager scores: {counts['scores']}; trophy counts checked: {counts['trophies']}; margins checked: {counts['margins']}")
print('Bounds, boolean types, league-title derivation, distinct positions, bonus caps, totals and winner rules: '+('FAIL' if errors else 'PASS'))
print('Errors: '+str(len(errors)))
for error in errors:
    print('ERROR: '+error)
raise SystemExit(bool(errors))
```
